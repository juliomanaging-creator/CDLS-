## FRAUD DETECTION MECHANISMS

### Multi-Layered Approach

**Layer 1: Statistical Anomaly Detection**
```python
def detect_statistical_anomalies(transactions):
    """
    Identifies outliers using multiple methods
    """
    # Method 1: Z-score (standard deviations from mean)
    z_scores = stats.zscore(transactions['amount'])
    zscore_outliers = transactions[abs(z_scores) > 3]
    
    # Method 2: IQR (Interquartile Range)
    Q1 = transactions['amount'].quantile(0.25)
    Q3 = transactions['amount'].quantile(0.75)
    IQR = Q3 - Q1
    iqr_outliers = transactions[
        (transactions['amount'] < Q1 - 1.5*IQR) | 
        (transactions['amount'] > Q3 + 1.5*IQR)
    ]
    
    # Method 3: Benford's Law (first digit distribution)
    first_digits = transactions['amount'].astype(str).str[0].astype(int)
    benford_expected = [30.1, 17.6, 12.5, 9.7, 7.9, 6.7, 5.8, 5.1, 4.6]
    benford_actual = first_digits.value_counts(normalize=True) * 100
    benford_violation = chi_square_test(benford_actual, benford_expected)
    
    return {
        'zscore_outliers': zscore_outliers,
        'iqr_outliers': iqr_outliers,
        'benford_violation': benford_violation
    }
```

```python
    """
    Machine learning-based fraud detection
    """
    # Feature engineering
    features = pd.DataFrame({
        'amount': transactions['amount'],
        'hour_of_day': transactions['timestamp'].dt.hour,
        'day_of_week': transactions['timestamp'].dt.dayofweek,
        'days_since_last_txn': calculate_days_since_last(transactions),
        'amount_zscore': zscore(transactions['amount']),
        'vendor_frequency': calculate_vendor_frequency(transactions),
        'approver_frequency': calculate_approver_frequency(transactions),
        'weekend_transaction': transactions['timestamp'].dt.dayofweek >= 5,
        'after_hours': (transactions['timestamp'].dt.hour < 7) | 
                      (transactions['timestamp'].dt.hour > 18)
    })
    
    # Isolation Forest (unsupervised anomaly detection)
    iso_forest = IsolationForest(contamination=0.01, random_state=42)
    anomaly_predictions = iso_forest.fit_predict(features)
    
    # Random Forest (supervised classification if labeled data exists)
    if 'fraud_label' in transactions.columns:
        rf_model = RandomForest(n_estimators=100, random_state=42)
        rf_model.fit(features, transactions['fraud_label'])
        fraud_probability = rf_model.predict_proba(features)[:, 1]
    else:
        fraud_probability = None
    
    return {
        'anomaly_scores': iso_forest.score_samples(features),
        'fraud_probability': fraud_probability,
        'high_risk_transactions': features[anomaly_predictions == -1]
    }
```

**Layer 4: Network Analysis**
```python
import networkx as nx

def detect_collusion_networks(transactions):
    """
    Identifies suspicious vendor-employee networks
    """
    # Build bipartite graph: employees <-> vendors
    G = nx.Graph()
    
    for _, txn in transactions.iterrows():
        G.add_edge(
            f"EMP_{txn['approver_id']}", 
            f"VEN_{txn['vendor_id']}",
            weight=txn['amount']
        )
    
    # Detect communities (potential collusion rings)
    communities = nx.community.louvain_communities(G)
    
    # Flag suspicious patterns
    suspicious = []
    for community in communities:
        employees = [n for n in community if n.startswith('EMP_')]
        vendors = [n for n in community if n.startswith('VEN_')]
        
        if len(employees) > 1 and len(vendors) > 1:
            # Multiple employees working with same vendors
            suspicious.append({
                'employees': employees,
                'vendors': vendors,
                'total_amount': sum_community_transactions(G, community),
                'risk_score': calculate_network_risk(G, community)
            })
    
    return suspicious
```

```python
from web3 import Web3
import hashlib
import json

class BlockchainAuditAnchor:
    """
    Anchors audit data to Ethereum for immutability
    """
    
    def __init__(self, rpc_url, contract_address):
        self.w3 = Web3(Web3.HTTPProvider(rpc_url))
        self.contract = self.w3.eth.contract(
            address=contract_address,
            abi=AUDIT_CONTRACT_ABI
        )
    
    def anchor_audit_report(self, report_data):
        """
        Create immutable record of audit report
        """
        # Generate merkle tree of all findings
        findings_hashes = [
            hashlib.sha256(json.dumps(finding).encode()).hexdigest()
            for finding in report_data['findings']
        ]
        
        merkle_root = self.compute_merkle_root(findings_hashes)
        
        # Prepare metadata
        metadata = {
            'report_id': report_data['report_id'],
            'dept_id': report_data['dept_id'],
            'report_date': report_data['report_date'].isoformat(),
            'auditor': report_data['auditor'],
            'findings_count': len(report_data['findings']),
            'risk_level': report_data['risk_level']
        }
        
        # Submit to blockchain
        tx_hash = self.contract.functions.anchorAuditReport(
            merkle_root,
            json.dumps(metadata)
        ).transact({'from': self.w3.eth.defaultAccount})
        
        # Wait for confirmation
        receipt = self.w3.eth.wait_for_transaction_receipt(tx_hash)
        
        return {
            'merkle_root': merkle_root,
            'tx_hash': receipt.transactionHash.hex(),
            'block_number': receipt.blockNumber,
            'timestamp': self.w3.eth.get_block(receipt.blockNumber).timestamp
        }
    
    def verify_audit_integrity(self, report_id, merkle_root):
        """
        Verify report hasn't been tampered with
        """
        on_chain_data = self.contract.functions.getAuditReport(report_id).call()
        
        return {
            'is_valid': on_chain_data['merkleRoot'] == merkle_root,
            'on_chain_root': on_chain_data['merkleRoot'],
            'provided_root': merkle_root,
            'block_number': on_chain_data['blockNumber'],
            'timestamp': on_chain_data['timestamp']
        }
```

```python
def calculate_department_risk_score(dept_data):
    """
    Comprehensive risk scoring for state departments
    """
    
    # Component 1: Financial Risk (40% weight)
    financial_risk = (
        dept_data['budget_variance'] * 0.3 +
        dept_data['transaction_anomaly_rate'] * 0.3 +
        dept_data['procurement_compliance'] * 0.2 +
        dept_data['audit_finding_severity'] * 0.2
    )
    
    # Component 2: Operational Risk (30% weight)
    operational_risk = (
        dept_data['policy_violation_rate'] * 0.4 +
        dept_data['employee_turnover'] * 0.2 +
        dept_data['service_delivery_variance'] * 0.2 +
        dept_data['technology_maturity'] * 0.2
    )
    
    # Component 3: Compliance Risk (20% weight)
    compliance_risk = (
        dept_data['regulatory_violations'] * 0.5 +
        dept_data['legal_actions_pending'] * 0.3 +
        dept_data['whistleblower_complaints'] * 0.2
    )
    
    # Component 4: Reputational Risk (10% weight)
    reputational_risk = (
        dept_data['media_negative_coverage'] * 0.4 +
        dept_data['legislative_scrutiny'] * 0.3 +
        dept_data['public_complaints'] * 0.3
    )
    
    # Composite risk score
    composite_risk = (
        financial_risk * 0.40 +
        operational_risk * 0.30 +
        compliance_risk * 0.20 +
        reputational_risk * 0.10
    )
    
    # Risk level categorization
    if composite_risk >= 0.75:
        risk_level = 'CRITICAL'
        audit_frequency = 'weekly'
    elif composite_risk >= 0.50:
        risk_level = 'HIGH'
        audit_frequency = 'bi-weekly'
    elif composite_risk >= 0.25:
        risk_level = 'MEDIUM'
        audit_frequency = 'monthly'
    else:
        risk_level = 'LOW'
        audit_frequency = 'quarterly'
    
    return {
        'composite_risk_score': composite_risk,
        'risk_level': risk_level,
        'recommended_audit_frequency': audit_frequency,
        'component_scores': {
            'financial': financial_risk,
            'operational': operational_risk,
            'compliance': compliance_risk,
            'reputational': reputational_risk
        }
    }
```
