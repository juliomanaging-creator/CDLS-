CDLS Platform — File Inclusion Attack Patch

BASE_DIR_SNIPPET = '''
import os as _os

def _safe_path(user_path: str, base_dir: str = None) -> str:
    """Resolve and validate a user-supplied path against the allowed base directory."""
    if base_dir is None:
        base_dir = str(Path(__file__).resolve().parent)
    real = _os.path.realpath(_os.path.abspath(user_path))
    allowed = _os.path.realpath(_os.path.abspath(base_dir))

$sriPatch = @'
"""
CDLS Platform — SRI Integrity Hash Generator & Injector
Fetches external CDN scripts/styles and computes SHA-384 integrity hashes.
Injects integrity and crossorigin="anonymous" attributes into HTML files.
Run: python patch_sri_hashes.py
"""
import hashlib, base64, re, urllib.request, shutil
from pathlib import Path

CDN_ASSETS = [
    ("https://cdn.tailwindcss.com", "script", "tailwindcss"),
    ("https://cdn.jsdelivr.net/npm/@tensorflow/tfjs@4.17.0/dist/tf.min.js", "script", "tfjs"),
    ("https://cdn.jsdelivr.net/npm/@tensorflow-models/coco-ssd@2.2.3/dist/coco-ssd.min.js", "script", "coco-ssd"),
]

def sha384_of_url(url):
    try:
        req = urllib.request.Request(url, headers={"User-Agent": "CDLS-SRI-Bot/1.0"})
        with urllib.request.urlopen(req, timeout=30) as r:
            content = r.read()
        digest = hashlib.sha384(content).digest()
        return "sha384-" + base64.b64encode(digest).decode()
    except Exception as e:
        return f"ERROR: {e}"

hashes = {}
print("Computing SRI hashes (this may take 30–60 seconds)...")
for url, tag, name in CDN_ASSETS:
    print(f"  Fetching: {name}...", end="", flush=True)
    h = sha384_of_url(url)
    hashes[url] = h
    print(f" {h[:20]}...")

html_files = list(Path(".").glob("*.html")) + list(Path(".").glob("**/*.html"))
for html_file in html_files:
    shutil.copy(html_file, html_file.with_suffix(".html.sri.bak"))
    content = html_file.read_text(encoding="utf-8")
    changed = False
    for url, h in hashes.items():
        if "ERROR" in h:
            print(f"[SKIP] Could not fetch {url}: {h}")
            continue
        # Match <script src="url"> or <link href="url"> without integrity attr
        pattern = rf'(<(?:script|link)[^>]+(?:src|href)=["\']){re.escape(url)}(["\'][^>]*)(>)'

            if "integrity=" in m.group(0): return m.group(0)
            return f'{m.group(1)}{url}{m.group(2)} integrity="{h}" crossorigin="anonymous"{m.group(3)}'
        new_content = re.sub(pattern, add_integrity, content, flags=re.IGNORECASE)
        if new_content != content:
            content = new_content
            changed = True
    if changed:
        html_file.write_text(content, encoding="utf-8")
        print(f"[FIXED] {html_file}: SRI hashes injected")
    else:

    print(f"  {url}")
    print(f"  → {h}")
    print()
