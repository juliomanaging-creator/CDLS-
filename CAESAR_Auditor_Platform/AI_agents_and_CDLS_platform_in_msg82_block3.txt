cd ~/Downloads

cd rd_tracker_automation

# 2. Open Chrome and go to:
chrome://extensions/

# 4. Click "Load unpacked"

# 5. Select: rd_tracker_automation/browser_extension

# 6. Extension installed! Pin it to toolbar for easy access
